Site Intelligence v4.39.0 R1 — Mac Terminal Commands

Copy only the commands below into Terminal. Do not copy this heading.

cd ~/Downloads

BUNDLE="$(find . -maxdepth 1 -type f \
  -name 'sustainable-catalyst-site-intelligence-v4.39.0-r1-release-bundle*.zip' \
  -print | sort | tail -1)"

[ -n "$BUNDLE" ] || {
  echo "ERROR: v4.39.0 R1 release bundle not found in Downloads."
  exit 1
}

RELEASE_DIR="$(mktemp -d "$HOME/Downloads/site-intelligence-v4.39.0-r1-release.XXXXXX")"

unzip -q "$BUNDLE" -d "$RELEASE_DIR"
cd "$RELEASE_DIR"

chmod +x install_and_push_site_intelligence_v4_39_0_r1_macos.sh

REPO_DIR="$HOME/Downloads/sustainable-catalyst-site-intelligence"

if [ ! -d "$REPO_DIR/.git" ]; then
  git clone \
    https://github.com/Content-Catalyst-LLC/sustainable-catalyst-site-intelligence.git \
    "$REPO_DIR"
fi

./install_and_push_site_intelligence_v4_39_0_r1_macos.sh "$REPO_DIR"

echo
echo "WORDPRESS ZIP:"
echo "$RELEASE_DIR/sustainable-catalyst-site-intelligence-v4.39.0-r1-wordpress-plugin.zip"
echo
echo "Upload that ZIP in WordPress Plugins and choose Replace current with uploaded."
echo "The displayed plugin version remains 4.39.0; no VPS deployment is required."
