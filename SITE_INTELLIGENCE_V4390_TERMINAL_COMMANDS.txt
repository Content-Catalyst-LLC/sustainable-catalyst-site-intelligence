Site Intelligence v4.39.0 — Mac Terminal Commands

cd ~/Downloads

BUNDLE="$(find . -maxdepth 1 -type f \
  -name 'sustainable-catalyst-site-intelligence-v4.39.0-release-bundle*.zip' \
  -print | sort | tail -1)"

[ -n "$BUNDLE" ] || {
  echo "ERROR: v4.39.0 release bundle not found in Downloads."
  exit 1
}

rm -rf sustainable-catalyst-site-intelligence-v4.39.0-release
mkdir -p sustainable-catalyst-site-intelligence-v4.39.0-release

unzip -q "$BUNDLE" \
  -d sustainable-catalyst-site-intelligence-v4.39.0-release

cd sustainable-catalyst-site-intelligence-v4.39.0-release
chmod +x install_and_push_site_intelligence_v4_39_0_macos.sh

REPO_DIR="$HOME/Downloads/sustainable-catalyst-site-intelligence"

if [ ! -d "$REPO_DIR/.git" ]; then
  git clone \
    https://github.com/Content-Catalyst-LLC/sustainable-catalyst-site-intelligence.git \
    "$REPO_DIR"
fi

./install_and_push_site_intelligence_v4_39_0_macos.sh \
  "$REPO_DIR"
