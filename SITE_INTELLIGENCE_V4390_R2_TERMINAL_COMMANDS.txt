Site Intelligence v4.39.0 R2 — Mac and VPS Terminal Commands

MAC — copy only the commands below into Terminal.

cd ~/Downloads

BUNDLE="$(find . -maxdepth 1 -type f \
  -name 'sustainable-catalyst-site-intelligence-v4.39.0-r2-release-bundle*.zip' \
  -print | sort | tail -1)"

[ -n "$BUNDLE" ] || {
  echo "ERROR: v4.39.0 R2 release bundle not found in Downloads."
  exit 1
}

RELEASE_DIR="$(mktemp -d "$HOME/Downloads/site-intelligence-v4.39.0-r2-release.XXXXXX")"
unzip -q "$BUNDLE" -d "$RELEASE_DIR"
cd "$RELEASE_DIR"
chmod +x install_and_push_site_intelligence_v4_39_0_r2_macos.sh

REPO_DIR="$HOME/Downloads/sustainable-catalyst-site-intelligence"

if [ ! -d "$REPO_DIR/.git" ]; then
  git clone \
    https://github.com/Content-Catalyst-LLC/sustainable-catalyst-site-intelligence.git \
    "$REPO_DIR"
fi

./install_and_push_site_intelligence_v4_39_0_r2_macos.sh "$REPO_DIR"

echo
echo "WORDPRESS ZIP:"
echo "$RELEASE_DIR/sustainable-catalyst-site-intelligence-v4.39.0-r2-wordpress-plugin.zip"
echo
echo "HOMEPAGE SOURCE:"
echo "$RELEASE_DIR/sustainable-catalyst-homepage-v4.3.1.html"
